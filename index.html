function doGet(e) {
  var action = e.parameter.action;
  if (action === "getUsers") return responseJSON(getUsersData());
  if (action === "getMessages") return responseJSON(getMessagesData());
  return responseJSON({ status: "success", message: "API Active" });
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action;

    if (action === "sendMessage") {
      return responseJSON(saveMessage(contents.data));
    } else if (action === "saveUser") {
      return responseJSON(saveUser(contents.data));
    }
    return responseJSON({ status: "error", message: "Invalid Action" });
  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() });
  }
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function getUsersData() {
  var sheet = getOrCreateSheet("Users", ["id", "username", "password", "name", "role", "status"]);
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var headers = data[0];
  return data.slice(1).map(function(row) {
    var obj = {};
    headers.forEach(function(h, i) { 
      var key = h;
      if (h === "username") key = "email";
      obj[key] = row[i]; 
    });
    return obj;
  });
}

function getMessagesData() {
  var sheet = getOrCreateSheet("Messages", ["id", "fromId", "told", "text", "timestamp"]);
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var headers = data[0];
  return data.slice(1).map(function(row) {
    var obj = {};
    headers.forEach(function(h, i) { 
      var key = h;
      if (h === "fromId") key = "senderId";
      if (h === "told") key = "receiverId";
      obj[key] = row[i]; 
    });
    return obj;
  });
}

function saveMessage(msg) {
  var sheet = getOrCreateSheet("Messages", ["id", "fromId", "told", "text", "timestamp"]);
  sheet.appendRow([msg.id, msg.senderId, msg.receiverId, msg.text, msg.timestamp]);
  return { status: "success", data: msg };
}

function saveUser(user) {
  var sheet = getOrCreateSheet("Users", ["id", "username", "password", "name", "role", "status"]);
  var data = sheet.getDataRange().getValues();
  var foundIndex = -1;
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === user.id || data[i][1] === user.email) {
      foundIndex = i + 1;
      break;
    }
  }
  if (foundIndex > 0) {
    sheet.getRange(foundIndex, 1, 1, 6).setValues([[user.id, user.email, user.password, user.name, user.role, user.status]]);
  } else {
    sheet.appendRow([user.id, user.email, user.password, user.name, user.role, user.status]);
  }
  return { status: "success", data: user };
}

function getOrCreateSheet(sheetName, headers) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(headers);
  }
  return sheet;
}
